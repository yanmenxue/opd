C. 路由重标定（CPU，~1 分钟）

这是最关键的一条。重点看 step 级的 corr(disp, steplen)（路由实际用的就是步内均值），以及换 window 后是否变化：

$PY - <<'EOF'
import json, math
from gam_opsd.route import dispersion

def pearson(xs, ys):
  n = len(xs); mx = sum(xs)/n; my = sum(ys)/n
  sxx = sum((x-mx)**2 for x in xs); syy = sum((y-my)**2 for y in ys)
  sxy = sum((x-mx)*(y-my) for x, y in zip(xs, ys))
  return sxy/math.sqrt(sxx*syy) if sxx > 0 and syy > 0 else float('nan')

def step_corr(disp, spans):
  ds = [sum(disp[s:e])/(e-s) for s, e in spans]; ls = [e-s for s, e in spans]
  return pearson(ds, ls) if len(ds) > 1 else float('nan')

def tok_corr(disp, spans):
  st = [e-s for s, e in spans for _ in range(e-s)]
  return pearson(disp, st)

def avg(v):
  v = [x for x in v if x == x]
  return sum(v)/len(v) if v else float('nan')

for rd in ['round_000', 'round_009', 'round_019']:
  rows = [json.loads(l) for l in open('runs/main/%s/graded.jsonl' % rd, encoding='utf-8')]
  for w in [8, 16, 32, 64]:
      T, S = [], []
      for r in rows:
          for x in r['responses']:
              a = x['a']
              if len(a) < 4: continue
              sp = x.get('step_spans') or [(0, len(a))]
              d = dispersion(a, window=w)
              T.append(tok_corr(d, sp)); S.append(step_corr(d, sp))
      print("%s w=%-3d  token级 %+.3f   <<step级 %+.3f>>" % (rd, w, avg(T), avg(S)))
  print()

# τ 的标定 + 分位数路由的比例
rows = [json.loads(l) for l in open('runs/main/round_019/graded.jsonl', encoding='utf-8')]
means = []
for r in rows:
  for x in r['responses']:
      a = x['a']
      if len(a) < 4: continue
      sp = x.get('step_spans') or [(0, len(a))]
      d = dispersion(a, window=16)
      for s, e in sp: means.append(sum(d[s:e])/(e-s))
ms = sorted(means); N = len(ms)
print("步骤总数 %d   步内 mean Disp 分位：" % N)
for p in [0.1, 0.25, 0.5, 0.75, 0.9]:
  print("  p%-4.2f = %.3f" % (p, ms[min(N-1, int(p*N))]))
EOF

判读：

- step 级 corr 明显 > 0 而 token 级 ≈ 0 → 说明 spec 的前提是成立的，只是用错了层级（该按步内均值相关，不该按 token）→ 直接写进论文，并解释之前 -0.036 是层级错配
- 两个都 ≈ 0 → 前提本身不成立，GAM 的立论要重写（这比调参重要得多）
- 窗宽 w 从 8→64 相关性变化很大 → 说明估计器不稳，需要报告稳健性

D. 长度实探测（GPU，~10 分钟）

决定 max_new_tokens/max_seq_len。给足预算（8192）跑 20 条，看模型自然写多长、在哪里 box、有没有  thinking：

$PY - <<'EOF'
import statistics, sys
sys.path.insert(0, '.')
from gam_opsd.common import build_llm, chat_many, load_config
from gam_opsd.data import load_eval
from gam_opsd.prompts import qa_messages

MODEL = '/disk1/duhw/models/ms_qwen3.5-4b/Qwen/Qwen3.5-4B'
cfg = load_config('configs/base.yaml')
rows = load_eval(cfg)[:20]
llm = build_llm(MODEL, tp=1, max_model_len=10240)
outs = chat_many(llm, [qa_messages(r['question']) for r in rows],
               n=1, temperature=0.0, top_p=1.0, max_tokens=8192)
L = []; nb = 0; nt = 0
for r, comps in zip(rows, outs):
  c = comps[0]; t = c.text; L.append(len(c.token_ids))
  hb = '\\boxed' in t
  ht = (' thinking' in t) or ('<|thinking|>' in t)
  nb += hb; nt += ht
  print("%-10s len=%5d boxed=%d think=%d tail=%r"
        % (r['qid'], len(c.token_ids), hb, ht, t[-100:]))
s = sorted(L)
print("\n长度 min=%d p25=%d median=%d p75=%d max=%d  mean=%.0f"
    % (s[0], s[len(s)//4], statistics.median(s), s[3*len(s)//4], s[-1], statistics.mean(s)))
print("触顶(==8192) %d/%d   boxed率 %.2f   thinking标记率 %.2f"
    % (sum(x >= 8190 for x in L), len(L), nb/len(L), nt/len(L)))
