  1. 权重差 —— 判定训练到底推动了多少（最关键）

  $PY - <<'EOF'
  from pathlib import Path
  from safetensors import safe_open
  BASE=Path('/disk1/duhw/models/ms_qwen3.5-4b/Qwen/Qwen3.5-4B')
  CKPT=Path('ckpt/model.safetensors')
  bp={}
  for p in sorted(BASE.glob('*.safetensors')):
      with safe_open(p,framework='pt') as f:
          for k in f.keys():
              if k.startswith('model.language_model.'):
                  bp[k[len('model.language_model.'):]]=p
  sf=safe_open(CKPT,framework='pt'); sk=list(sf.keys())
  print("base 文本张量:",len(bp)," ckpt 张量:",len(sk))
  print("ckpt 样例:",sk[:2]); print("base 样例:",list(bp)[:2])
  rows=[]
  for k in sk:
      if k not in bp: continue
      with safe_open(bp[k],framework='pt') as f: b=f.get_tensor('model.language_model.'+k).float()
      s=sf.get_tensor(k).float(); scale=b.abs().mean().item()+1e-12
      rows.append(((b-s).abs().max().item()/scale,k))
  rows.sort(reverse=True)
  print("对齐 %d/%d 张量，平均相对变化 %.3e" % (len(rows),len(sk),sum(r for r,_ in rows)/max(len(rows),1)))
  for r,k in rows[:12]: print("  %.3e  %s" % (r,k))
  EOF

  判读：平均相对变化在 1e-6 量级 = 权重基本没动（就是没训起来）；1e-3 以上 = 确实在学。这个数比 loss 可靠得多。

  3. 优势量级分解 + 路由分布

  $PY - <<'EOF'
  import json
  from gam_opsd.route import gam_advantage
  rows=[json.loads(l) for l in open('runs/main/round_019/graded.jsonl',encoding='utf-8')]
  infos=[]; A=[]; C=[]
  for row in rows:
      for resp in row['responses']:
          a=resp.get('a') or []
          if not a: continue
          _,_,info=gam_advantage(a, resp.get('step_spans'), mode='dynamic', tau=0.5, window=16,
              eps=1e-6, w_kl=1.0, lambda_step=1.0, adv_norm=False,
              correct=resp.get('correct',False), correct_scale=1.0,
              acc_adv=resp.get('acc_adv'), lambda_acc=1.0)
          infos.append(info)
          A.append(sum(abs(x) for x in a)/len(a))
          C.append(abs(resp.get('acc_adv') or 0.0))
  agg=lambda k:(lambda v: sum(v)/len(v) if v else float('nan'))([i[k] for i in infos if i.get(k) is not None])
  for k in ['disp_mean','disp_median','disp_p25','disp_p75',
            'frac_step_tokens','frac_steps_routed','n_steps','corr_disp_steplen']:
      print("%-20s %.4f" % (k, agg(k)))
  print("%-20s %.5f   <- 蒸馏项 w_kl·|a_t|" % ('mean|a_t|', sum(A)/len(A)))
  print("%-20s %.5f   <- RL 项 |acc_adv|"  % ('mean|acc_adv|', sum(C)/len(C)))
  print("RL / 蒸馏 幅度比 ≈ %.0fx" % (sum(C)/max(sum(A),1e-9)))
  print("acc_adv==0 的响应占比 %.1f%%  (组内 8 条同对/同错)" % (100*sum(x==0 for x in C)/len(C)))
  print("响应数 %d" % len(infos))
  EOF

  disp_p25/p75、frac_steps_routed 是 trainer 算了但没打印的三个数，判断 τ 该怎么定就看它们。
