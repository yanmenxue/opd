from pathlib import Path
from safetensors import safe_open
BASE=Path('/disk1/duhw/models/ms_qwen3.5-4b/Qwen/Qwen3.5-4B')

  rows.append(((s-b).abs().max().item()/sc,(s-b).abs().mean().item()/sc,k))
rows.sort(reverse=True)
print("对齐 %d/%d  平均相对变化 max=%.3e mean=%.3e" % (

for a,m,k in rows[:15]: print("  max %.3e  mean %.3e  %s" % (a,m,k))
print("未对齐:", [k for k in sk if k not in bp])
